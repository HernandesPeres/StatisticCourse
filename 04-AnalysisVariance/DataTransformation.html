# QQ Plot
qqnorm(data$resp)
qqline(data$resp, col=2)

# Box-Cox estimation
bc <- boxcox(resp ~ trat, data = data, plotit = TRUE)

lambda <- bc$x[which.max(bc$y)]

data$resp_boxcox <- if (lambda == 0) log(data$resp) else (data$resp^lambda - 1)/lambda
data_long <- data %>%
  pivot_longer(cols = c(resp, resp_boxcox), names_to = "method", values_to = "values")

data_long <- data_long %>%
  group_by(method) %>%
  mutate(scaled_values = scale(values)) %>%
  ungroup()

ggplot(data_long, aes(x = scaled_values, fill = method)) +
  geom_density(alpha = 0.4, position = "identity") +
  labs(x = "Scaled values",
       y = "Freq.") +
  theme_classic() +
  scale_fill_manual(values = c("red", "limegreen"), labels = c("Before", "After"))+
  theme(legend.title = element_blank(),
        legend.position = "bottom")
